\section{Network}

\subsection*{Aim}
One network, three feature arms:
\begin{itemize}
  \item \textbf{PH arm}: persistence images, one per (filtration, homology dimension)
  \item \textbf{PersLay arm}: the same diagrams, vectorised by the network instead of rasterised
  \item \textbf{Classical arm}: sampled summary functions, $L,F,G,J$
\end{itemize}
Two tasks:
\begin{itemize}
  \item \textbf{Classify}: 5-way family (Poisson, Thomas, nested, Mat\'ern II, LGCP)
  \item \textbf{Params}: the family's own parameters, one model per family
\end{itemize}
Everything after the feature step is shared, so the arms are comparable by construction.

\subsection*{Inputs}
\begin{itemize}
  \item \textbf{Images}: $64\times64$ (or just $64$ for Rips and alpha $H_0$)
  \item \textbf{Diagrams} (PersLay): padded to a fixed length, columns (birth, persistence, mass)
  \item \textbf{Curves}: 512 samples each
  \item \textbf{Covariate}: $\log n$, the realised count, one scalar
\end{itemize}

\subsection*{Architecture (\texttt{PHNet})}
\begin{itemize}
  \item \textbf{Encoder}, one per input key, 1-D or 2-D following the input's rank:
  \begin{align*}
    [\text{conv}_3\,32 \to \text{ReLU} \to \text{avgpool}_2 \to \text{drop}]
    & \to [\text{conv}_3\,64 \to \text{ReLU} \to \text{avgpool}_2 \to \text{drop}] \\
    & \to [\text{conv}_3\,128 \to \text{ReLU}] \\
    & \to \text{adaptive avgpool}(4) \\
    & \to \text{flatten} \\
    & \to \text{Linear} \\
    & \to \text{ReLU},
  \end{align*}
  embedding dimension 64. Padding 1, stride 1, so the convolutions keep resolution and only the pools shrink it: $64^2\to32^2\to16^2\to4^2$ in 2-D, $512\to\dots\to4$ for a curve
  \item \textbf{Head}: concat(embeddings, $\log n$) $\to$ MLP $[64,32]$, ReLU, dropout 0.1 $\to$ 5 logits, or $p$ targets
\end{itemize}

\subsection*{PersLay encoder}
A diagram is a \emph{set}, so any vectorisation of it is permutation invariant, and PersLay
(Carri\`ere et al.\ 2020) writes every such map in one form,
\[
  V(D) \;=\; \mathrm{op}\bigl(\{\, w(p)\,\varphi(p) : p \in D \,\}\bigr),
\]
with a point transformation $\varphi$, a scalar weight $w$ and a permutation-invariant pooling
$\mathrm{op}$. The persistence image is the frozen member of that family: $\varphi$ a Gaussian on a
fixed grid of centres, $w(p)$ the linear weight (the persistence), $\mathrm{op}$ the sum. This arm
keeps that shape and \emph{learns} it:
\begin{itemize}
  \item $\varphi$: $Q=64$ Gaussians with learned centres and bandwidths, in the standardised
        (birth, persistence) plane the imager also works in; centres initialised spread over
        $[-2,2]^2$, bandwidth $4/\sqrt{Q}$
  \item $w$: $\mathrm{softplus}(\mathrm{MLP}_{2\to16\to1}(p))$, non-negative because a weight is a
        mass, multiplied by the point's own mass ($1/n$ under density scaling), so the $n$-scaling
        of the image arm carries over unchanged
  \item $\mathrm{op}$: sum (max and mean are available), then dropout, Linear, ReLU to the same
        embedding dimension 64 the image encoder produces
\end{itemize}
Capacity is fitted like the imager's box: it holds every point of 99\% of the \emph{training}
diagrams, capped at 1024, and a longer diagram keeps its most persistent points --- the ones the
image's linear weight would have made brightest. Padded rows carry mass exactly $0$, so padding is
invisible to the pooling.

\subsection*{How inputs combine}
\begin{itemize}
  \item $H_0$ and $H_1$: separate encoders, embeddings concatenated
  \item Several filtrations (multi-$k$): one \emph{shared-weight} encoder applied to each in turn, embeddings concatenated
  \item Curves on one grid: stacked as channels of the first convolution, since index $i$ is then the same $r$ in every channel and the convolution can compare them there
  \item Curves on different grids: one encoder each, meeting only after pooling
\end{itemize}

\subsection*{Design choices}
\begin{itemize}
  \item Pool to 4 instead of global. A conv stack under a global average is translation-invariant for \emph{every} weight setting; in a persistence image position \emph{is} the measurement. One blob at two positions: relative embedding distance $0.0002$ under a global pool, $0.14$ at pool 4
  \item Channel-wise dropout (Dropout1d/2d, $p=0.2$): neighbouring pixels are correlated, so element-wise dropout regularises little
  \item No normalisation layers anywhere (no BatchNorm, no LayerNorm). All normalisation is on the input and target side, where it is explicit and fitted on train rows only
\end{itemize}

\section{Normalisation of covariates and targets}
\begin{itemize}
  \item $\log n$, $z$-scored on train
  \item Targets: $\log$ then $z$-score per target. Loss is MSE on the standardised log targets, so each target is weighted equally whatever its actual scale
  \item Predictions are inverted ($\cdot\,$sd $+$ mean, $\exp$) and saved in original parameter units
  \item Per family: Poisson $\nb$; Thomas $\kappa,\mu,\sigma$; nested $\kappa,\mu_1,\mu_2,\sigma_1,\sigma_2$; Mat\'ern $R,\lambda_p$; LGCP $\nb,\sigma^2,s$ ($\nb$ in place of $\mu_{\log}$, the only target that can be negative)
\end{itemize}

\section{Evaluation}

\subsection*{Split}
\begin{itemize}
  \item By $\theta$ index, not by pattern: train $\theta<7000$, val $7000\le\theta<8000$, test $8000\le\theta<10\,000$
  \item Replicates of a $\theta$ fall in the same split, so no parameter vector is seen in both train and test
  \item $\theta_i$ is drawn from its own stream, so indices are i.i.d.\ prior draws and a contiguous block is a random split
  \item The same split for every family, task, arm and seed
\end{itemize}

\subsection*{Training}
\begin{itemize}
  \item AdamW, lr $10^{-3}$, weight decay $3\cdot10^{-4}$, batch 128
  \item Loss: cross-entropy (classify), MSE on standardised log targets (params)
  \item Up to 200 epochs; early stopping on val loss, patience 30; best-val weights restored
  \item Val selects the stopping epoch and nothing else
  \item The seed sets network init, minibatch order and dropout. Features and split are identical across seeds, so seeds measure optimisation variance, not data variance
\end{itemize}

\subsection*{Metrics}
\begin{itemize}
  \item Every test pattern's prediction is saved with its \texttt{case\_id} and joined to the manifest for $(\nb,\dt)$ afterwards, so the binning can change without retraining
  \item \textbf{Classify}: accuracy, and NLL of the true family. Restricted to one family, accuracy is that family's recall
  \item \textbf{Params}: RMSE in log units, per target and pooled over targets
  \item \textbf{Cells}: 8 log-$\dt$ bins on $[0.1,4]$ $\times$ 3 log-$\nb$ bins on $[100,800]$, plus every marginal. Edges come from the prior and are shared by every family and method; cells with fewer than 10 test $\theta$ are skipped
  \item Poisson has $\dt=0$ and so falls in no $\dt$ bin: it appears in the marginal and $\nb$ rows only. A $\dt$ cell of the classification task therefore holds four families, and chance in it is $0.25$
\end{itemize}
