\section*{1. What changed}

\textbf{Cloud bank} (replaces the $\dt$-sweep)
\begin{itemize}
  \item $50\,000$ $\theta$ $\times$ 2 patterns $\times$ 5 families $=500$k clouds (was $10$k $\theta$); $\nb\sim$ log-U$[75,950]$
  \item Drawn on \emph{physical} parameters, two independent dials: richness (points/cluster) and smearing $\omega=\sigma\sqrt\kappa\in[0.2,8]$
  \item $\dt$ is now a post-hoc \emph{label}: refitting the null tables relabels, never re-simulates
  \item Bounds inverted into the draw's support, not rejected: 0 rejections (LGCP $1.7\%$, numerical grid limit). Old design: nested rejected $18\%$ and tilted the prior
  \item $\mathrm{cv}(n)\le0.30$ (was $0.45$): the label is evaluated at $\nb$ but describes realised $n$; drift $\approx$ linear in cv
  \item Test set: $12\,000$ $\theta$/family (was $2\,000$), so near-CSR cells are populated
\end{itemize}

\textbf{Regime analysis}
\begin{itemize}
  \item $\dt$ reduction \texttt{sup}$\to$\texttt{ext}: extremum of $\hat L-r$ \emph{before} studentising. No radius window, exact size ($0.0501$). Fixes Mat\'ern hard cores below $r_{\min}$ (power $19\%\to81\%$ at $\nb=200$)
  \item Unclamped: $\dt=0$ is CSR's noise floor, Poisson sits at $-0.97$; $\dt=1$ is the $5\%$ rejection boundary
  \item Signed arms $\dt_{\text{lo}}$ (repulsion), $\dt_{\text{hi}}$ (clustering) stored as alternative coordinates
  \item 10 geometric $\dt$ bins on $[0.05,32]$ $\times$ 3 $\nb$ bins; bootstrap over test $\theta$; paired differences
  \item Undetectable share ($\dt<1$): Thomas $55\%$, LGCP $52\%$, Mat\'ern $29\%$, nested $20\%$
  \item Below the first bin (unbinned, marginal only): Thomas $42\%$, LGCP $39\%$, nested $13\%$, Mat\'ern $10\%$
\end{itemize}

\textbf{PersLay} (third arm)
\begin{itemize}
  \item Same diagrams, learned vectorisation $V(D)=\mathrm{op}\{w(p)\varphi(p)\}$, $Q=64$, sum pooling, $\approx4.4$k parameters per dim
  \item Variants: plain, and \texttt{z} (pooled vector z-scored)
  \item Run: classification only (plain 10 seeds, \texttt{z} 2 seeds so far). \textbf{Not yet run for parameters}
\end{itemize}
