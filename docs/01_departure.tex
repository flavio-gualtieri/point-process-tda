\section{Theory}

\subsection*{Setting}
\begin{itemize}
  \item Window $W=[0,1]^2$, fixed
  \item $\nb$: expected number of points (deterministic model parameter)
  \item Pattern $x = \{x_1,\dots,x_n\} \subset W$; $n = |x|$ realised number of points (random around $\nb$)
  \begin{itemize}
    \item Poisson, Mat\'ern II: $n/\nb \in [0.90, 1.11]$ (5--95\%)
    \item Thomas, nested, LGCP: $n/\nb \in [0.77, 1.25]$ (overdispersed, clusters arrive whole)
  \end{itemize}
\end{itemize}

\subsection*{Estimator}
\begin{itemize}
  \item Ripley's $K$, isotropic edge correction, $\lambda^2$ estimated by $n(n-1)$:
  \[
    \hat K_x(r) = \frac{1}{n(n-1)} \sum_{i\neq j} w(x_i, d_{ij})\, \mathbf 1\{d_{ij}\le r\},
    \qquad
    w(x,d) = \frac{2\pi}{\text{length of } \partial b(x,d)\cap W}
  \]
  \item Centred $L$-function of a pattern $x$, and its model counterpart:
  \[
    \hat\phi_x(r) = \hat L_x(r) - r = \sqrt{\hat K_x(r)/\pi} - r,
    \qquad
    \phi(r) = L(r) - r \quad (= 0 \text{ under CSR})
  \]
  \item Evaluated on the radius grid: $\hat\phi_x = \big(\hat\phi_x(r_1),\dots,\hat\phi_x(r_{512})\big) \in \R^{512}$
  \item Radii $r_k = 0.25\,k/512$, $k=1,\dots,512$ ($r=0$ dropped: identically zero)
\end{itemize}

\subsection*{Radius set}
Everything in this subsection is defined for fixed $n$. The test and $\dt$ only use radii in $R(n)$.
\begin{itemize}
  \item At small $r$, $\hat\phi_x(r)$ is a function of a small count: the pairs of $x$ closer than $r$. Under $\mathbb P_{0,n}$ we expect
  \[
    \mu(r;n) = \tfrac12 n(n-1)\pi r^2
  \]
  such pairs (edge-corrected)
  \item The maximum in $M_n$ (next subsection) treats all radii alike only if each standardised deviation $(\hat\phi_X(r) - m_0)/s_0$ has roughly the same null distribution, i.e.\ $\approx N(0,1)$
  \item Few expected pairs $\Rightarrow$ $\hat\phi$ takes a handful of values; standardised deviation discrete and skewed, not $N(0,1)$. Checked on the null curves at $n=99$:
  \begin{itemize}
    \item $\mu < 0.3$: heavy right tail, $\mathbb P(|z|>3)$ 10$\times$ Gaussian
    \item $\mu \in [3, 10)$: still 3$\times$ Gaussian
    \item $\mu \ge 10$: Gaussian tail rate, skewness and kurtosis
  \end{itemize}
  \item Heavy-tailed radii win the null maximum too often $\Rightarrow$ $c_{95}$ inflated by counting noise $\Rightarrow$ power lost at every other radius ($c_{95}$ at $n=99$: 3.07 with $p=10$, 3.43 with $p=0.1$)
  \item Keep radii with at least $p$ expected pairs, $p = 10$ (usual threshold for the normal approximation to a Poisson count):
  \[
    R(n) = \{r_k : \mu(r_k;n) \ge p\} = [r_{\min}(n), 0.25],
    \qquad
    r_{\min}(n) = \sqrt{\frac{2p}{\pi n(n-1)}} \approx \frac{2.52}{n}
  \]
  \item $r_{\min}$: 0.13 ($n=20$), 0.025 ($n=100$), 0.0034 ($n=800$), 0.0015 ($n=2000$). More points $\Rightarrow$ $R(n)$ reaches down to finer scales
\end{itemize}

\subsection*{Null distribution}
Everything in this subsection is defined for fixed $n$. In practice, we compute the null distribution at every $n$ on a grid.
\begin{itemize}
  \item Null $\mathbb P_{0,n}$: $n$ i.i.d.\ uniform points on $W$ (CSR conditioned on $n$); $\mathbb E_{0,n}$, $\mathrm{sd}_{0,n}$ under it
  \item Generate $12\,000$ CSR clouds from $\mathbb P_{0,n}$
  \item Estimate null moments for $r \in R(n)$ using half of the CSR clouds:
  \[
    m_0(r;n) = \mathbb E_{0,n}\big[\hat\phi_X(r)\big],
    \qquad
    s_0(r;n) = \mathrm{sd}_{0,n}\big[\hat\phi_X(r)\big]
  \]
  \item Using remaining half of CSR clouds, compute maximum deviation of each cloud $x$:
  \[
    M_n(x) = \max_{r\in R(n)} \frac{|\hat\phi_x(r) - m_0(r;n)|}{s_0(r;n)}
  \]
  \item Determine critical value: $c_{95}(n)$ = 95\% quantile of $M_n$ over these clouds (estimates the 95\% quantile of $M_n(X)$, $X\sim\mathbb P_{0,n}$)
\end{itemize}

\subsection*{Test (studentised global envelope; Myllym\"aki et al.\ 2017)}
\[
  S(x) = \frac{M_n(x)}{c_{95}(n)}, \qquad n = |x|
\]
\begin{itemize}
  \item Under the null, $\mathbb P_{0,n}(S > 1) = 0.05$ by definition of $c_{95}$ (up to Monte Carlo error)
  \item Reject CSR at 5\% iff $S > 1$
\end{itemize}

\subsection*{Distance $\dt$}
\[
  \dt(\theta,\nb) = \frac{1}{c_{95}(\nb)} \max_{r\in R(\nb)} \frac{|L_\theta(r) - r|}{s_0(r;\nb)}
\]
\begin{itemize}
  \item Same functional as $S$, applied to the closed-form $L_\theta$ at $\nb$, uncentred ($m_0 = 0$)
  \item $\dt(\text{CSR}) = 0$ exactly
  \item $\dt = 1$: true curve on the rejection boundary $\Rightarrow$ power well above 5\%; measured 0.64--0.78 (02), not 50\%, because pattern noise adds to the maximum
  \item Units: null s.d.s, relative to the critical value
  \item Property of the model, known before simulation $\Rightarrow$ usable as a sweep coordinate
  \item At fixed shape, $\dt$ grows with $\nb$ ($s_0$ shrinks)
\end{itemize}

\subsection*{$\dt$ does not transfer well for Mat\'ern II}
\begin{itemize}
  \item Hard core $R$: no two points closer than $R$ $\Rightarrow$ $L(r) - r = -r$ for $r < R$
  \item Size of the hard core in pairs: the CSR pairs it forbids,
  \[
    \mu_R = \mu(R;\nb) = \tfrac12 \nb(\nb-1)\pi R^2 = p\,\big(R/r_{\min}(\nb)\big)^2
  \]
  \item For $\dt \lesssim 1$ the hard core lies below $r_{\min}$ ($\mu_R < p$), where $\dt$ does not look. $\dt$ sees only the deficit it leaves above $r_{\min}$, which decays like $1/r$ $\Rightarrow$ maximum at $r_{\min}$ for 82\% of Mat\'ern $\theta$ $\Rightarrow$ Mat\'ern $\dt$ is set by the cutoff $p$, not by the model
  \item A nearest-neighbour test sees the hard core directly: reject CSR if no pair is closer than $d$, with $\mu(d) \approx 3$ ($\mathbb P_0(\text{no pair within } d) \approx e^{-\mu(d)} = 0.05$; 2.97 measured)
  \item Same $\dt$, very different difficulty (all 20\,000 simulated Mat\'ern patterns; nearest-neighbour test calibrated to 5\% on the Poisson pool):

  \smallskip
  \begin{tabular}{@{}lrrrrrr@{}}
  \toprule
  $\dt$ bin & (0.1, 0.2] & (0.2, 0.3] & (0.3, 0.5] & (0.5, 1] & (1, 2] & (2, 4] \\
  \midrule
  $L$ test power & 0.06 & 0.07 & 0.12 & 0.34 & 0.86 & 1.00 \\
  Nearest-neighbour test power & 0.23 & 0.63 & 0.97 & 1.00 & 1.00 & 1.00 \\
  \bottomrule
  \end{tabular}
  \item So $\dt$ is a fair difficulty axis only for departures visible inside $R(\nb)$ (the cluster families). For Mat\'ern we use $\mu_R$ as the regime coordinate:
  \begin{itemize}
    \item Unit: pairs removed; detectable by the nearest-neighbour test from $\mu_R \approx 3$ on
    \item Same ordering as $\dt$ (Spearman 0.999) and free of $\nb$: at fixed $\mu_R$, $L$-test power flat across $\nb$ terciles while $R$ varies $4\times$
    \item Sweep unchanged: Mat\'ern still drawn on $\dt \in [0.1, 4]$, which induces $\mu_R \in [1, 46]$
    \item Mat\'ern not pooled with the cluster families in $\dt$-binned results
  \end{itemize}
\end{itemize}

\subsection*{Smoothing in $n$}
\begin{itemize}
  \item True $m_0$, $s_0$, $c_{95}$ smooth in $n$; raw per-$n$ estimates noisy
  \item Dense grid alone does not help: leave-one-out interpolation error on the old 20-point grid $\approx$ Monte Carlo noise
  \item Fit across $n$ $\Rightarrow$ borrow strength from neighbouring $n$; defined at every $n$, no interpolation rule
  \item $s_0 \propto n^{-1}$ at small $r$, slower at large $r$ $\Rightarrow$ fit $n\,m_0$, $n\,s_0$ (slowly varying)
  \item Cubic B-splines in $\log n$
\end{itemize}

\begin{itemize}
  \item $n$ grid: $\mathrm{round}(20 \cdot 1.03^k)$ up to 2000 $\Rightarrow$ 153 values
  \item Per grid $n$: 6000 fit + 6000 calibration patterns
  \item Validation: 50\,000 patterns, $n$ log-uniform integer on $[20, 2000]$
  \item $p = 10$ pairs; $\alpha = 0.05$; knots: 8 interior (moments), 4 interior ($c_{95}$)
\end{itemize}

\subsection*{Steps}
\begin{enumerate}
  \item \textbf{Simulate} (\texttt{simulate --jobs 10}; $\sim$55 min on 10 cores)
  \begin{itemize}
    \item Seed key: (\texttt{pilot}, \texttt{poisson}, $n\cdot 10^5 + \text{rep}$, PATTERN)
    \item Validation key: (\texttt{pilot}, \texttt{poisson}, $i < 10^5$); $n$ from PARAMS, points from PATTERN
    \item One \texttt{float32} file per $n$ ($12000 \times 512$), atomic, resumable; 3.6\,GB total
  \end{itemize}
  \item \textbf{Fit} (\texttt{fit})
  \begin{itemize}
    \item Per $n$, fit batch: pointwise mean, s.d., Monte Carlo s.e.s
    \item Weighted least squares per radius: $n\,m_0$, $n\,s_0$ on spline basis in $\log n$
    \item Per $n$, calibration batch: max statistic under \emph{fitted} moments $\to$ raw $c_{95}$, s.e.
    \item Weighted spline fit of $c_{95}$ in $\log n$
    \item Store knots + coefficients (100\,KB)
  \end{itemize}
  \item \textbf{Validate} (\texttt{validate}): rejection rate on validation patterns, overall and in 6 $\log n$ bands
  \item \textbf{Use}: \texttt{Tables().statistic(curves, n)} (test), \texttt{Tables().delta\_tilde(L\_minus\_r, nbar)} ($\dt$)
\end{enumerate}

\section{Statistical notes}

\subsection*{Sample splitting}
\begin{itemize}
  \item Three disjoint sets: fit ($m_0$, $s_0$), calibration ($c_{95}$), validation (size)
  \item $c_{95}$ from same patterns as moments $\Rightarrow$ downward-biased critical value
  \item Size on calibration batch is 5\% by construction $\Rightarrow$ validation needs fresh patterns
  \item $c_{95}$ computed under the \emph{fitted} moments $\Rightarrow$ calibrates the statistic actually used
  \item Validation $n$ random integers $\Rightarrow$ mostly off-grid, tests the smoother between grid points
\end{itemize}

\subsection*{Fit diagnostics}
\begin{itemize}
  \item $z = (\text{raw} - \text{fit})/\text{s.e.}$; pure noise $\Rightarrow$ mean $z^2 \approx 1 - \text{params}/\text{grid points}$
  \item s.e.\ of $s_0$ uses the fourth moment: $\sqrt{(\mu_4 - s^4)/N}/(2s)$ (small-$r$ tails non-Gaussian)
  \item $c_{95}$ s.e.\ from order statistics: $(q_{.96}-q_{.94})/0.02 \cdot \sqrt{.95\cdot.05/N}$
  \item Per-$n$ mean $z^2$ varies widely (0.07--4.5): residuals strongly correlated across $r$ $\Rightarrow$ few effective d.o.f.\ per $n$; band averages $\approx 1$, no trend
\end{itemize}

\subsection*{Centring}
\begin{itemize}
  \item $\hat K$ unbiased under binomial; $\hat L = \sqrt{\hat K/\pi}$ is not (Jensen) $\Rightarrow$ $m_0 \ne 0$
  \item $\dt$ uncentred, test centred $\Rightarrow$ gap $\le 0.03$ in $\dt$ units on $R(n)$
  \item Old tables ($r \ge 0.01$ for all $n$): gap 0.16 at $n=50$ --- caused by low-pair radii
\end{itemize}

\subsection*{Interpretation of $\dt$}
\begin{itemize}
  \item Defined through a classical L-test $\Rightarrow$ yardstick of that test, not an omnibus distance
  \item Classical baseline partly calibrated to the axis: state explicitly
  \item Sup over $r$: one radius decides; multi-scale models (nested Thomas) summarised by their most visible scale
  \item $L$ only: hard cores below $r_{\min}$ are invisible to $\dt$ (Mat\'ern II, above)
  \item Design $\dt$ at $\nb$, not realised $n$: per-pattern difficulty scatters around the label (most for Cox processes)
  \item $\dt=1$ at $\nb$ does not mean $S=1$ on a pattern: $S$ is random around $\dt$
\end{itemize}

\subsection*{Scope and limits}
\begin{itemize}
  \item Tables valid for $n \in [20, 2000]$; outside raises (simulated pool: $n$ = 44--961)
  \item $W$ fixed unit square; weight formula requires $r \le 0.5$
  \item $R(n)$ depends on $n$: at small $n$ small-scale structure invisible by construction
  \item Curves stored $\Rightarrow$ $p$, $\alpha$, knots changeable without re-simulation
\end{itemize}
